cmake_minimum_required(VERSION 3.25)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" DEFAULT_VERSION LIMIT_COUNT 1)
set(GAMEGAUGE_VERSION "${DEFAULT_VERSION}" CACHE STRING "Release version")
set(GAMEGAUGE_REPOSITORY "" CACHE STRING "GitHub owner/repository")
if(NOT GAMEGAUGE_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "Release version must be major.minor.patch")
endif()
if(GAMEGAUGE_REPOSITORY AND NOT GAMEGAUGE_REPOSITORY MATCHES "^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$")
  message(FATAL_ERROR "Invalid GitHub repository")
endif()
project(GameGauge VERSION ${GAMEGAUGE_VERSION} LANGUAGES CXX)
configure_file(src/common/version.h.in generated/version.h @ONLY)
configure_file(src/common/version_resource.rc.in generated/version_resource.rc @ONLY)
include_directories("${CMAKE_BINARY_DIR}/generated")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT WIN32)
  message(FATAL_ERROR "GameGauge requires Windows")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc)
endif()
add_library(gauge_common STATIC
  src/common/config.cpp src/common/frame_statistics.cpp src/common/history.cpp src/common/metric_style.cpp src/common/platform.cpp src/common/sample_validation.cpp src/common/update_release.cpp)
target_include_directories(gauge_common PUBLIC src third_party)
target_link_libraries(gauge_common PUBLIC shell32 advapi32)
add_library(gauge_metrics STATIC
  src/metrics/cpu_temperature.cpp src/metrics/disk.cpp src/metrics/hardware.cpp src/metrics/nvidia.cpp src/metrics/presentmon.cpp
  src/metrics/sampler.cpp src/metrics/target.cpp)
target_include_directories(gauge_metrics PRIVATE third_party/presentmon)
target_link_libraries(gauge_metrics PUBLIC gauge_common dxgi pdh psapi powrprof dwmapi setupapi)
add_executable(GameGauge WIN32
  src/host/main.cpp src/host/application.cpp src/host/ipc_server.cpp src/host/updater.cpp src/host/obs.cpp
  src/hud/overlay.cpp src/hud/renderer.cpp src/host/app.rc)
target_link_libraries(GameGauge PRIVATE gauge_metrics d3d11 d2d1 dwrite dcomp dwmapi comctl32 winhttp bcrypt crypt32)
target_link_options(GameGauge PRIVATE /MANIFEST:NO)
add_executable(GameGauge.Diagnostics src/diagnostics/main.cpp src/host/ipc_server.cpp)
target_link_libraries(GameGauge.Diagnostics PRIVATE gauge_metrics)
add_executable(GameGauge.CpuProbe src/diagnostics/cpu_temperature_probe.cpp)
target_include_directories(GameGauge.CpuProbe PRIVATE src)
target_link_libraries(GameGauge.CpuProbe PRIVATE bcrypt advapi32)
add_executable(GameGauge.Settings WIN32
  src/settings/main.cpp src/settings/window.cpp src/settings/controls.cpp src/settings/window_chrome.cpp src/settings/pages.cpp src/settings/simple_pages.cpp src/settings/history_page.cpp src/host/ipc_server.cpp src/hud/renderer.cpp src/host/app.rc)
target_link_libraries(GameGauge.Settings PRIVATE gauge_common d2d1 dwrite windowscodecs comdlg32 d3d11 dcomp dwmapi)
target_link_options(GameGauge.Settings PRIVATE /MANIFEST:NO)
add_executable(GameGauge.RenderProbe WIN32 tests/render_probe.cpp)
target_link_libraries(GameGauge.RenderProbe PRIVATE d3d11 dxgi)
add_executable(GameGauge.Tests tests/core_tests.cpp tests/telemetry_regressions.cpp tests/metric_style_tests.cpp)
target_link_libraries(GameGauge.Tests PRIVATE gauge_metrics)
add_executable(GameGauge.IpcTests tests/ipc_tests.cpp src/host/ipc_server.cpp)
target_link_libraries(GameGauge.IpcTests PRIVATE gauge_common)
enable_testing()
add_library(gauge_setup_ui STATIC src/setup/window.cpp)
target_link_libraries(gauge_setup_ui PUBLIC d2d1 dwrite windowscodecs dwmapi)
add_executable(GameGauge.SetupPreview WIN32 src/setup/preview.cpp src/host/app.rc)
target_link_libraries(GameGauge.SetupPreview PRIVATE gauge_setup_ui shell32)
target_link_options(GameGauge.SetupPreview PRIVATE /MANIFEST:NO)
if(EXISTS "${CMAKE_SOURCE_DIR}/build/setup/payload.rc")
  add_executable(GameGauge.Setup WIN32 src/setup/main.cpp "${CMAKE_SOURCE_DIR}/build/setup/payload.rc")
  target_link_libraries(GameGauge.Setup PRIVATE gauge_setup_ui shell32)
  target_link_options(GameGauge.Setup PRIVATE /MANIFEST:NO)
  set_target_properties(GameGauge.Setup PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
add_test(NAME core_contracts COMMAND GameGauge.Tests)
add_test(NAME ipc_contracts COMMAND GameGauge.IpcTests)
set_target_properties(GameGauge GameGauge.Diagnostics GameGauge.CpuProbe GameGauge.Settings GameGauge.RenderProbe GameGauge.Tests GameGauge.IpcTests GameGauge.SetupPreview
  PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
